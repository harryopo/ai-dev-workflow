```python
# ============================================================
# v6.15 引擎 kwargs 构造（X-D16）
# ============================================================

def _build_engine_kwargs(args, search_depth: str = '') -> Dict[str, Any]:
    """把编排层参数转成引擎 kwargs。

    X-D16：include_domains / exclude_domains / days / time_range 在
    engines/mcp_engines.py:158-165 与 engines/fallback.py:653 早已被消费，
    但编排层从不传值——参数收得到、永远拿不到。这里补上传参路径。

    铁律：**只在显式指定时才塞键**。传空值与不传对引擎语义不同
    （如 time_range='' 可能被当成"不限时间"而非"用引擎默认值"），
    所以未指定的键一律不出现，让引擎保持自己的默认行为。
    """
    kw: Dict[str, Any] = {}
    if search_depth:
        kw['search_depth'] = search_depth
    tr = str(getattr(args, 'time_range', '') or '').strip()
    if tr:
        kw['time_range'] = tr
    inc = [d.strip() for d in (getattr(args, 'include_domain', None) or [])
           if str(d).strip()]
    if inc:
        kw['include_domains'] = inc
    exc = [d.strip() for d in (getattr(args, 'exclude_domain', None) or [])
           if str(d).strip()]
    if exc:
        kw['exclude_domains'] = exc
    return kw
```

- [ ] **Step 3: 实现 `_resolve_search_depth`**

```python
# effort/depth 档位 → Tavily 检索深度。键是 plan.resolve_preset_key()
# 归一**之后**的 preset key（注意 exhaustive→extreme）。
SEARCH_DEPTH_BY_PRESET = {
    'quick': 'fast',
    'standard': 'basic',
    'deep': 'advanced',
    'extreme': 'advanced',
}


def _resolve_search_depth(args, explicit: Optional[str] = None) -> str:
    """决定本次检索下发给 Tavily 的 search_depth。

    X-D17：此前 research.py 全文 search_depth 出现 0 次，主入口所有调研
    一律让 Tavily 走 basic——关键子问题（定量数据、争议核对）拿不到
    advanced 的跨索引重排。这里把 effort 档位接到 search_depth 上。

    显式值优先；未指定时按档位推导，让默认行为随档位变好而不是靠用户记得手填。
    """
    if explicit:
        return explicit
    try:
        from plan import resolve_preset_key
        key = resolve_preset_key(getattr(args, 'effort', None),
                                 getattr(args, 'depth', None))
    except Exception:
        key = 'standard'
    return SEARCH_DEPTH_BY_PRESET.get(key, 'basic')
```

```python
    'quick': 3,
    'standard': 4,
    'deep': 5,
    'extreme': 5,
}

# 扇出顺序即优先级：先补最缺的（反方 → 量化 → 原始 → 最新 → 基础）
FACET_ORDER = ('counter', 'data', 'primary', 'fresh', 'fact')

# 引擎失败兜底里，查询生成层根本不知道能用算子。

def build_query_facets(subtopic: str, preset_key: str = 'standard') -> List[Dict[str, Any]]:
    """为一个子问题生成类型化的检索查询集合（X-D18 + X-D20）。

    Args:
        subtopic: 子问题文本
        preset_key: resolve_preset_key() 归一后的档位键

    Returns:
        [{'facet': 'counter', 'query': '...', 'cures': '...'}]
    """
    base = (subtopic or '').strip()
    if not base:
        return []
    n = FACET_COUNTS.get(preset_key, FACET_COUNTS['standard'])
    out: List[Dict[str, Any]] = []

        out.append({

            'query': query,
            'cures': spec['cures'],
        })
    return out
```

```python
# ============================================================
# v6.15 高风险 claim 规则筛查（X-D21）
# ============================================================
#
# 现有链路评的是"这个源可不可信"（CRAAP 五维 + Tier 分级），
# 没有一步评"这句话是否真被这段文字支持"。Tier 高、CRAAP 高的来源，
# 配上被过度引申的 claim，照样全绿过门。
#
# **本函数只筛出待查清单，不做验证**。真要判定"支持/不支持"需要 LLM，
# 那会让校验门从确定性脚本变成依赖模型调用的不确定环节，
# 并破坏 --stamp 的可复现性——本轮明确不做。
# 漏检率不可控，SKILL.md 与 CHANGELOG 须如实声明，不得宣称"已实现支持度回查"。

import re as _re

# 定量：数字 + 单位/量度。只看数字会误伤"2026 年"这类中性时间表述，
# 因此要求数字与单位/量度词共现。
HIGH_RISK_PATTERNS = {
    'quantitative': [
        _re.compile(r'\d+(?:\.\d+)?\s*(?:%|％|倍|万|亿|千|ms|毫秒|秒|分钟|小时|天|GB|MB|KB|TB|QPS|RPS|TPS)'),
        _re.compile(r'\d+(?:\.\d+)?\s*(?:万|亿)\s*(?:用户|次|条|个|家)'),
    ],
    'absolute': [
        _re.compile(r'(?:全部|所有|任何|总是|从不|永远|唯一|必然|彻底|绝对)'),
        _re.compile(r'(?:第一|最好|最强|最快|最高|最佳|最先进)'),
    ],
    'comparative': [
        _re.compile(r'(?:优于|胜过|远超|领先于|比.{0,12}更)'),
        _re.compile(r'(?:领先|优势明显|碾压|吊打)'),
    ],
}

HIGH_RISK_LABELS = {
    'quantitative': '定量断言（数字+单位）',
    'absolute': '绝对化断言（全部/唯一/最…）',
    'comparative': '比较级断言（优于/远超…）',
}


def screen_high_risk_claims(claims: List[Dict[str, Any]],
                            report=None) -> List[Dict[str, Any]]:
    """筛出定量/绝对化/比较级三类高风险 claim。

    命中**不降级、不删除**，只标记为 needs_support_check 交 Lead 在
    Phase 3.5 逐条定夺。理由与"不可信标记只降级不删除"同理：
    攻击者或噪声故意植入特征词来躲避收录，删除即变成 DoS-by-poisoning。

    Args:
        claims: 账本 claim 列表（每项须含 id 与 text）
        report: 可选 ValidationReport，写入 stats 与 warnings

    Returns:
        [{'claim_id', 'text', 'kind', 'label', 'matched'}]
    """
    hits: List[Dict[str, Any]] = []
    for c in claims or []:
        text = str(c.get('text', '') or '')
        if not text:
            continue
        for kind, pats in HIGH_RISK_PATTERNS.items():
            for pat in pats:
                m = pat.search(text)
                if m:
                    hits.append({
                        'claim_id': c.get('id'),
                        'text': text,
                        'kind': kind,
                        'label': HIGH_RISK_LABELS[kind],
                        'matched': m.group(0),
                    })
                    break   # 同一 claim 同一类只记一次，跨类可重复
    if report is not None:
        report.stats['needs_support_check'] = len(hits)
        if hits:
            preview = '、'.join(str(h.get('claim_id')) for h in hits[:8])
            report.warnings.append(
                f'{len(hits)} 条高风险 claim 需人工确认来源是否真的支持该表述'
                f'（{preview}）——这是待查清单，不是已判定不支持；'
                f'请在 Phase 3.5 逐条核对正文与来源，必要时改写或降级')
    return hits
```

## 安全专项说明（实现者必读）

用户明确要求"特别注意安全问题"。本轮涉及的安全点：

1. **X-D19 选 A 而非 B 是安全决策，不是省事**。B 方案会把注入面放大一个数量级
   （全文比摘要长约 100 倍，可藏载荷位置成倍增加），与安全文档 S.6 的硬约束冲突。
   实现时**不得**"顺手"改成 B 方案。
2. **X-D21 是账本污染防护**。它对应安全文档里的攻击目标 3——
   "不改控制流，只让虚假 claim 通过溯源校验，产出看起来严谨实则被投毒的报告"。
   实现时**不得**把它改成自动降级或自动删除（会变成 DoS-by-poisoning）。
3. **不得引入 LLM 语义判定**。会让校验门从确定性脚本变成不确定环节，
   且破坏 `--stamp` 的可复现性——那是 v6.11 建立的防伪机制。
4. **X-D18 扇出增加来源数 = 增加注入暴露面**。这一点必须在 CHANGELOG 与 SKILL.md 记录，
   不得只写收益不写代价。
5. **本轮不实现 L1 出口白名单 / L2 结构化隔离 / L3 不可信标记 / L4 canary**——
   它们是安全文档 S.5 的四道防线，属独立立项。不得声称本轮已有任何注入防护。
